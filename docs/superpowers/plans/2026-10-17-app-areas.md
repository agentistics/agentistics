# APP.AREAS — navigation by areas · Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Dashboard|Sessions switch and the flat link list with one area tree (desktop), an icon rail + sessions column (Sessions area), and a place-pill + bottom sheet (mobile), and add the Uso and Liderança pages.

**Architecture:** One pure registry (`lib/appAreas.ts`) describes every area, subpage, route, icon and label; a pure resolver maps a URL to `{area, page}`. Every surface (tree, rail, mobile sheet, page header, search) renders from it. New pages are thin compositions of components that already exist; the only server change is exposing data `/api/plan-limits` already computes.

**Tech Stack:** React 19 + Vite, react-router-dom 7, bun:test (no jsdom — pure tests and `renderToStaticMarkup`), lucide-react icons, Bun server.

**Spec:** `docs/superpowers/specs/2026-10-17-app-areas.md` (read it first). Prototypes: https://claude.ai/artifact/M35gYVX7Wjo3ydc2Q1z5zA (boards "Opção A — Uso › Agora", "A · Sessões — jeito 2", "Celular 2").

```ts
/**
 * appAreas.ts — the ONE description of the app's navigation.
 *
 * The desktop tree, the Sessions icon rail, the mobile sheet, the page header crumb and the page
 * search all render from this list. A second list anywhere is a second answer to "where is this
 * page", which is how the old SideNav `items` and the mobile `navTiles` drifted apart.
 *
 * Matching is by SEGMENT boundary (same rule the old `modeOfPath` had): `/sessions-report` is not the
 * Sessions area.
 */
export type AreaId = 'sessions' | 'tasks' | 'leadership' | 'usage' | 'metrics' | 'repositories' | 'vault'

/** lucide-react icon names; the components map them, so this module stays React-free. */
export type AreaIcon = 'terminal' | 'clipboard' | 'crown' | 'gauge' | 'chart' | 'git' | 'lock'

  path: string

  labelPt: string

  beta?: boolean

/** Routes that are deliberately in no area: redirects, the footer's settings, export. */
export const UNLISTED_ROUTES: readonly string[] = ['projects', 'hardware', 'settings', 'export']

function under(pathname: string, prefix: string): boolean {
  if (prefix === '/') return pathname === '/'
  return pathname === prefix || pathname.startsWith(prefix + '/')
}

/**
 * Which area and page a URL belongs to. The MOST SPECIFIC prefix wins, so `/tasks/activity` is the
 * activity page while `/tasks/t-123` is the board.
 */
export function areaOfPath(pathname: string): { area: Area; page: AreaPage | null } | null {
  let best: { area: Area; page: AreaPage | null; len: number } | null = null
  for (const area of APP_AREAS) {
    if (area.root && under(pathname, area.root)) {
      if (!best || area.root.length > best.len) best = { area, page: null, len: area.root.length }
    }
    for (const page of area.pages) {
      for (const prefix of [page.path, ...(page.also ?? [])]) {
        if (under(pathname, prefix) && (!best || prefix.length > best.len)) best = { area, page, len: prefix.length }
      }
    }
  }
  return best ? { area: best.area, page: best.page } : null
}

export function pageLabel(x: { labelPt: string; labelEn: string }, lang: 'pt' | 'en'): string {
  return lang === 'pt' ? x.labelPt : x.labelEn
}

const fold = (s: string) => s.normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().trim()

/** Pages whose label (either language) or area label contains the query. */
export function searchAreas(query: string, _lang: 'pt' | 'en'): { area: Area; page: AreaPage }[] {
  const q = fold(query)
  if (!q) return []
  const out: { area: Area; page: AreaPage }[] = []
  for (const area of APP_AREAS) {
    const pages: AreaPage[] = area.pages.length
      ? [...area.pages]
      : [{ id: area.id, path: area.root!, labelPt: area.labelPt, labelEn: area.labelEn }]
    for (const page of pages) {
      const hay = [page.labelPt, page.labelEn, area.labelPt, area.labelEn].map(fold)
      if (hay.some(h => h.includes(q))) out.push({ area, page })
    }
  }
  return out
}

/** Where clicking an area goes: the remembered page of THAT area, else its first page. */
export function firstPath(area: Area, remembered?: string | null): string {
  const r = (remembered ?? '').trim()
  if (r && areaOfPath(r)?.area.id === area.id) return r
  return area.pages[0]?.path ?? area.root ?? '/'
}
```

```tsx
/**
 * PageHeader — the one header every area page draws: crumb, title, ONE status sentence, ≤2 actions
 * (owner's page rules, 2026-10-17). The crumb is read from the registry so it can never disagree with
 * the tree.
 */
import { APP_AREAS, pageLabel, type AreaId } from '../../lib/appAreas'

export interface PageAction { label: string; onClick: () => void; primary?: boolean }

export function PageHeader({ lang, areaId, pageId, title, status, actions = [] }: {
  lang: 'pt' | 'en'; areaId: AreaId; pageId?: string; title?: string; status?: string; actions?: readonly PageAction[]
}) {
  const area = APP_AREAS.find(a => a.id === areaId)!
  const page = area.pages.find(p => p.id === pageId)
  const heading = title ?? (page ? pageLabel(page, lang) : pageLabel(area, lang))
  return (
    <header style={{ display: 'flex', alignItems: 'flex-start', gap: 16, marginBottom: 20, flexWrap: 'wrap' }}>
      <div style={{ flex: '1 1 280px', minWidth: 0 }}>
        <div style={{ fontSize: 12, color: 'var(--text-tertiary)', marginBottom: 2 }}>
          {pageLabel(area, lang)}{page ? <> <span aria-hidden>›</span> {pageLabel(page, lang)}</> : null}
        </div>
        <h1 style={{ margin: 0, fontSize: 22, fontWeight: 600, color: 'var(--text-primary)' }}>{heading}</h1>
        {status ? <p style={{ margin: '6px 0 0', fontSize: 14, color: 'var(--text-secondary)' }}>{status}</p> : null}
      </div>
      {actions.slice(0, 2).map(a => (
        <button key={a.label} type="button" onClick={a.onClick} className={a.primary ? 'ag-btn ag-btn-primary' : 'ag-btn'}>{a.label}</button>
      ))}
    </header>
  )
}
```

```ts
import type { PlanBudget, PlanPace } from '@agentistics/core'

export type UsageBudgetRow = PlanBudget & PlanPace & { account?: string; accountLabel?: string; pauseUntil?: string }
export interface PlanSessionsRow { account: string; harness: string; label: string; parallel: number | null; pauseUntil: string | null; reason: string }

const hhmm = (iso: string) => { const d = new Date(iso); return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}` }

/** The Uso › Agora status sentence. Null when there is nothing to summarise. */
export function usageHeadline(rows: readonly UsageBudgetRow[], now: number, lang: 'pt' | 'en'): string | null {
  if (!rows.length) return null
  const over = rows.filter(r => r.paceStatus === 'over' || r.status === 'hold')
  const pauses = over.map(r => r.pauseUntil).filter((p): p is string => !!p && Date.parse(p) > now).sort()
  const latest = pauses[pauses.length - 1]
  if (!over.length) return lang === 'pt' ? `Todos os ${rows.length} planos no ritmo.` : `All ${rows.length} plans on pace.`
  const head = lang === 'pt' ? `${over.length} de ${rows.length} planos acima do ritmo.` : `${over.length} of ${rows.length} plans over pace.`
  if (!latest) return head
  return head + (lang === 'pt' ? ` Não abra sessões novas até ${hhmm(latest)}.` : ` Do not open new sessions until ${hhmm(latest)}.`)
}

/**
 * How many sessions fit in parallel, per account, at the sustainable rate. `hours` is how long the
 * person wants to work; it only bounds the pause (a pause beyond the window is still reported).
 * Per-session burn is measured burn divided by live sessions; with no live session it is unknown.
 */
export function planSessions(rows: readonly UsageBudgetRow[], _hours: number, _now: number): PlanSessionsRow[] {
  return rows.map(r => {
    const label = r.accountLabel ?? r.account ?? r.harness
    const base = { account: r.account ?? r.harness, harness: r.harness, label, pauseUntil: r.pauseUntil ?? null }
    if (r.status === 'hold') return { ...base, parallel: 0, reason: 'hold' }
    const live = r.liveSessions?.length ?? 0
    if (!live || !(r.burnPctPerHour > 0)) return { ...base, parallel: null, reason: 'unmeasured' }
    const perSession = r.burnPctPerHour / live
    return { ...base, parallel: Math.max(0, Math.floor(r.sustainablePctPerHour / perSession)), reason: 'pace' }
  })
}
```
